他說要分成成兩側去組，然後就變這樣了，tip不會設，哈哈XD程式show一下，雖然不是我自己寫出來的

from controller import Robot
import math

robot = Robot()
timestep = int(robot.getBasicTimeStep())

# 馬達初始化
motor1 = robot.getDevice("t1")
motor2 = robot.getDevice("t2")
motor1.setPosition(float('inf'))
motor2.setPosition(float('inf'))
motor1.setVelocity(0.0)
motor2.setVelocity(0.0)

# 控制參數
amplitude = 0.5  # 弧度範圍
frequency = 0.5  # Hz

while robot.step(timestep) != -1:
    t = robot.getTime()
    angle1 = amplitude * math.sin(2 * math.pi * frequency * t)
    angle2 = amplitude * math.cos(2 * math.pi * frequency * t)
    motor1.setVelocity(angle1)
    motor2.setVelocity(angle2)
